5.2 Kafka Consumer Implementation

JAVA PurchaseOrderKafkaConsumer
@Component
public class PurchaseOrderKafkaConsumer {
    
    private static final Logger logger = 
        LoggerFactory.getLogger(PurchaseOrderKafkaConsumer.class);
    
    @Autowired
    private ProcurementService procurementService;
    
    @Autowired
    private AuditService auditService;
    
    @KafkaListener(
        topics = "SCM_PurchaseOrder_Inbound",
        groupId = "procurement-service-group",
        containerFactory = "kafkaListenerContainerFactory"
    )
    @Transactional
    public void processPurchaseOrderInbound(
        ConsumerRecord<String, String> record,
        Acknowledgment acknowledgment) {
        
        try {
            String correlationId = record.key();
            PurchaseOrderEvent event = 
                objectMapper.readValue(record.value(), PurchaseOrderEvent.class);
            
            logger.info("Processing PO inbound event: {}", correlationId);
            procurementService.processPOEvent(event, correlationId);
            acknowledgment.acknowledge();
        } catch (JsonProcessingException e) {
            logger.error("Failed to deserialize message", e);
            handleDeserializationError(record);
        } catch (Exception e) {
            logger.error("Error processing PO inbound event", e);
            sendToDLQ(record, e);
        }
    }

    @KafkaListener(
        topics = "SCM_Inventory_Updated",
        groupId = "procurement-service-group"
    )
    public void processInventoryUpdate(InventoryUpdateEvent event) {
        try {
            procurementService.updateInventoryCache(event);
        } catch (Exception e) {
            logger.error("Failed to update inventory cache", e);
        }
    }
}

5.3 Controller & REST API

JAVA PurchaseOrderController
@RestController
@RequestMapping("/api/v1")
@Slf4j
public class PurchaseOrderController {
    
    @Autowired
    private ProcurementService procurementService;
    
    @Autowired
    private AuditService auditService;
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    @PostMapping("/purchase-orders")
    @Transactional
    @PreAuthorize("hasAnyRole('PROCUREMENT_ADMIN', 'BUYER')")
    public ResponseEntity<?> createPurchaseOrder(
        @RequestBody @Valid PurchaseOrderRequest request,
        @RequestHeader(value = "X-Tenant-ID") String tenantId,
        @RequestHeader(value = "X-Request-ID", required = false) String requestId) {
    
        if (requestId == null) {
            requestId = UUID.randomUUID().toString();
        }
        
        try {
            validateTenant(tenantId);
            validateRequest(request);
            auditService.logInboundRequest(requestId, request, "PO_CREATE");
            
            PurchaseOrder po = procurementService.createPurchaseOrder(
                request,
                tenantId
            );
            
            PurchaseOrderResponse response = PurchaseOrderResponse.from(po);
            return ResponseEntity
                .status(HttpStatus.CREATED)
                .header("Location", "/api/v1/purchase-orders/" + po.getId())
                .header("X-Request-ID", requestId)
                .body(response);
        } catch (ValidationException e) {
            return buildErrorResponse(400, e.getMessage(), requestId);
        } catch (BusinessException e) {
            return buildErrorResponse(422, e.getMessage(), requestId);
        } catch (Exception e) {
            log.error("Unexpected error creating PO", e);
            return buildErrorResponse(500, "Internal server error", requestId);
        }
    }

    @GetMapping("/purchase-orders/{id}")
    @PreAuthorize("hasAnyRole('PROCUREMENT_ADMIN', 'BUYER', 'SUPPLIER')")
    public ResponseEntity<?> getPurchaseOrder(@PathVariable String id) {
        try {
            PurchaseOrderResponse cachedPO = 
                (PurchaseOrderResponse) redisTemplate.opsForValue()
                    .get("po:" + id);
            
            if (cachedPO != null) {
                return ResponseEntity.ok(cachedPO);
            }
            
            PurchaseOrder po = procurementService.getPurchaseOrder(id);
            redisTemplate.opsForValue().set(
                "po:" + id,
                po,
                Duration.ofMinutes(30)
            );
            
            return ResponseEntity.ok(PurchaseOrderResponse.from(po));
        } catch (NotFoundException e) {
            return buildErrorResponse(404, "Purchase order not found", id);
        }
    }
}
